import BasicUsage from '@/components/Alert/BasicUsage.tsx';
import CloseButton from '@/components/Alert/CloseButton.tsx';

## Usage

The `Alert` component is a `div` with `.uk-alert` class. Use it as any `div` element.

<BasicUsage client:load>
  ```jsx
  <Alert>
    Eu eu cillum duis in officia aliquip enim qui sunt officia eu pariatur nostrud elit esse ut.
  </Alert>
  ```
</BasicUsage>

## Close button

Add the [`Close`](/close) component with `.uk-alert-close` to function as `Alert`'s close button.

```jsx
<Alert>
  <Close className="uk-alert-close" />
</Alert>
```

<CloseButton client:load>
  ```jsx
  <Alert>
    <Close className="uk-alert-close" />
    Eu eu cillum duis in officia aliquip enim qui sunt officia eu pariatur nostrud elit esse ut
    officia sint cupidatat nostrud nulla do aliquip veniam in ut sint culpa.
  </Alert>
  ```
</CloseButton>
